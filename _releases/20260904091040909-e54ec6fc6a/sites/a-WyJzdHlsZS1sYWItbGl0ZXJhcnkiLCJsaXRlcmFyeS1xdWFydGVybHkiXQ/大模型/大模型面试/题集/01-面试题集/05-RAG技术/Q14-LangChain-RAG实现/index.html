## 问题
请使用 LangChain 实现一个完整的生产级 RAG 系统，要求包含：文档加载、切片、向量化、混合检索、Rerank 重排序、幻觉防护。请写出核心代码并解释设计思路。

## 答案

以下是一个完整的生产级 RAG 系统实现，包含所有关键环节：

```python
"""
生产级 RAG 系统完整实现
包含：文档处理、向量存储、混合检索、Rerank、幻觉防护
"""

import os
from typing import List
from langchain_community.document_loaders import (
    TextLoader, PyPDFLoader, UnstructuredMarkdownLoader
)
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain_chroma import Chroma
from langchain.retrievers import BM25Retriever, EnsembleRetriever
from langchain.retrievers.document_compressors import CrossEncoderReranker
from langchain_community.cross_encoders import HuggingFaceCrossEncoder
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough, RunnableParallel
from langchain_core.documents import Document

# ───────────────────────────────────────────────
# 配置
# ───────────────────────────────────────────────
class RAGConfig:
    """RAG 系统配置"""
    CHUNK_SIZE = 500          # 文档切片大小
    CHUNK_OVERLAP = 100       # 重叠大小
    VECTOR_DIMENSION = 1536   # Embedding 维度
    TOP_K_RETRIEVE = 20       # 初步检索数量
    TOP_K_RERANK = 5          # Rerank 后保留数量
    SIMILARITY_THRESHOLD = 0.6  # 相似度阈值
    TEMPERATURE = 0.1         # 生成温度（降低幻觉）
    RERANK_MODEL = "BAAI/bge-reranker-v2-m3"

# ───────────────────────────────────────────────
# 1. 文档处理模块
# ───────────────────────────────────────────────
class DocumentProcessor:
    """文档加载与切片"""
    
    def __init__(self, config: RAGConfig):
        self.config = config
        self.splitter = RecursiveCharacterTextSplitter(
            chunk_size=config.CHUNK_SIZE,
            chunk_overlap=config.CHUNK_OVERLAP,
            separators=["\n\n", "\n", "。", "！", "？", "；", " ", ""],
            length_function=len,
        )
    
    def load_file(self, file_path: str) -> List[Document]:
        """根据文件类型选择加载器"""
        ext = os.path.splitext(file_path)[1].lower()
        
        loaders = {
            ".txt": TextLoader,
            ".pdf": PyPDFLoader,
            ".md": UnstructuredMarkdownLoader,
        }
        
        loader_class = loaders.get(ext, TextLoader)
        loader = loader_class(file_path)
        return loader.load()
    
    def process(self, file_paths: List[str]) -> List[Document]:
        """加载并切片文档"""
        all_docs = []
        for path in file_paths:
            docs = self.load_file(path)
            all_docs.extend(docs)
        
        # 添加元数据（来源文件、页码等）
        for doc in all_docs:
            doc.metadata["source"] = doc.metadata.get("source", "unknown")
        
        chunks = self.splitter.split_documents(all_docs)
        print(f"文档处理完成：{len(all_docs)} 篇文档 → {len(chunks)} 个 chunk")
        return chunks

# ───────────────────────────────────────────────
# 2. 向量存储模块
# ───────────────────────────────────────────────
class VectorStore:
    """向量数据库管理"""
    
    def __init__(self, config: RAGConfig, persist_dir: str = "./vector_db"):
        self.config = config
        self.persist_dir = persist_dir
        self.embeddings = OpenAIEmbeddings(
            model="text-embedding-3-small"
        )
        self.vectorstore = None
    
    def create(self, documents: List[Document]):
        """创建向量数据库"""
        self.vectorstore = Chroma.from_documents(
            documents=documents,
            embedding=self.embeddings,
            persist_directory=self.persist_dir
        )
        print(f"向量数据库创建完成，存储于 {self.persist_dir}")
    
    def load(self):
        """加载已有向量数据库"""
        self.vectorstore = Chroma(
            persist_directory=self.persist_dir,
            embedding_function=self.embeddings
        )
    
    def as_retriever(self, k: int = 20):
        """返回检索器"""
        return self.vectorstore.as_retriever(
            search_type="similarity",
            search_kwargs={"k": k}
        )
    
    def similarity_search_with_score(self, query: str, k: int = 20):
        """带分数的相似度搜索"""
        return self.vectorstore.similarity_search_with_score(query, k=k)

# ───────────────────────────────────────────────
# 3. 混合检索 + Rerank 模块
# ───────────────────────────────────────────────
class HybridRetriever:
    """混合检索（向量 + BM25）+ Rerank"""
    
    def __init__(self, config: RAGConfig, documents: List[Document]):
        self.config = config
        
        # 向量检索器
        self.vector_store = VectorStore(config)
        self.vector_store.create(documents)
        vector_retriever = self.vector_store.as_retriever(k=config.TOP_K_RETRIEVE)
        
        # BM25 检索器
        bm25_retriever = BM25Retriever.from_documents(documents)
        bm25_retriever.k = config.TOP_K_RETRIEVE
        
        # 混合检索（Ensemble：加权融合）
        self.ensemble_retriever = EnsembleRetriever(
            retrievers=[bm25_retriever, vector_retriever],
            weights=[0.3, 0.7]  # BM25 权重 0.3，向量权重 0.7
        )
        
        # Rerank 模型
        cross_encoder = HuggingFaceCrossEncoder(model_name=config.RERANK_MODEL)
        self.reranker = CrossEncoderReranker(
            model=cross_encoder,
            top_n=config.TOP_K_RERANK
        )
    
    def retrieve(self, query: str) -> List[Document]:
        """执行混合检索 + Rerank"""
        # 第一步：混合检索
        initial_docs = self.ensemble_retriever.invoke(query)
        
        # 第二步：Rerank 精排
        reranked_docs = self.reranker.compress_documents(
            documents=initial_docs,
            query=query
        )
        
        return reranked_docs

# ───────────────────────────────────────────────
# 4. RAG 链组装
# ───────────────────────────────────────────────
class RAGSystem:
    """完整 RAG 系统"""
    
    def __init__(self, config: RAGConfig = None):
        self.config = config or RAGConfig()
        self.processor = DocumentProcessor(self.config)
        self.retriever = None
        self.chain = None
    
    def build_chain(self, documents: List[Document]):
        """构建 RAG 链"""
        # 初始化混合检索器
        self.retriever = HybridRetriever(self.config, documents)
        
        # LLM
        llm = ChatOpenAI(
            model="gpt-3.5-turbo",
            temperature=self.config.TEMPERATURE
        )
        
        # Prompt 模板（包含防幻觉约束）
        prompt = ChatPromptTemplate.from_messages([
            ("system", """你是一个严格基于参考资料回答问题的助手。
请遵守以下规则：
1. 只能基于提供的参考资料回答
2. 如果参考资料中没有相关信息，明确说"我不知道"
3. 回答时引用参考资料来源
4. 不要添加参考资料之外的信息
"""),
            ("human", """参考资料：
{context}

问题：{question}

请基于参考资料回答。""")
        ])
        
        # 格式化文档函数
        def format_docs(docs: List[Document]) -> str:
            return "\n\n---\n\n".join([
                f"[来源: {doc.metadata.get('source', 'unknown')}]\n{doc.page_content}"
                for doc in docs
            ])
        
        # LCEL 链组装
        self.chain = (
            RunnableParallel({
                "context": lambda x: format_docs(self.retriever.retrieve(x["question"])),
                "question": RunnablePassthrough()
            })
            | prompt
            | llm
            | StrOutputParser()
        )
    
    def ingest(self, file_paths: List[str]):
        """文档摄入"""
        documents = self.processor.process(file_paths)
        self.build_chain(documents)
        print("RAG 系统构建完成")
    
    def query(self, question: str) -> dict:
        """执行查询"""
        if self.chain is None:
            raise ValueError("请先执行文档摄入（ingest）")
        
        # 获取检索结果
        retrieved_docs = self.retriever.retrieve(question)
        
        # 相似度阈值检查（兜底）
        if len(retrieved_docs) == 0:
            return {
                "answer": "抱歉，未找到相关信息。",
                "sources": [],
                "has_answer": False
            }
        
        # 生成答案
        answer = self.chain.invoke({"question": question})
        
        return {
            "answer": answer,
            "sources": [
                {
                    "content": doc.page_content[:200],
                    "source": doc.metadata.get("source", "unknown")
                }
                for doc in retrieved_docs
            ],
            "has_answer": True
        }

# ───────────────────────────────────────────────
# 5. 使用示例
# ───────────────────────────────────────────────
def main():
    # 初始化系统
    rag = RAGSystem()
    
    # 文档摄入
    rag.ingest([
        "docs/product_manual.pdf",
        "docs/faq.md",
        "docs/policy.txt"
    ])
    
    # 查询
    result = rag.query("产品的退换货政策是什么？")
    
    print(f"\n答案：{result['answer']}")
    print(f"\n来源：")
    for source in result['sources']:
        print(f"  - {source['source']}: {source['content']}...")

if __name__ == "__main__":
    main()

系统设计思路：

模块 设计决策 理由
文档切片 RecursiveCharacterTextSplitter 优先保持语义完整性
向量存储 Chroma 开源、易用、足够大多数场景
混合检索 EnsembleRetriever（BM25 + 向量） 兼顾精确匹配和语义相似
Rerank BGE-Reranker-v2-m3 中文效果好，可本地部署
幻觉防护 Prompt 约束 + 相似度阈值 多层防护
链组装 LCEL 声明式、可观测、易维护

延伸追问

Q: 如何对这个 RAG 系统进行性能优化？

A: 1) 使用 GPU 加速 Embedding 和 Rerank；2) 对文档做预过滤（元数据过滤）；3) 使用更快的 LLM（如 GPT-3.5-turbo 替代 GPT-4）；4) 缓存热门查询结果；5) 异步处理文档摄入。

Q: 如何评估这个 RAG 系统的效果？

A: 使用 RAGAS 框架评估：Faithfulness（忠实度）、Answer Relevancy（相关性）、Context Precision（上下文精度）、Context Recall（上下文召回率）。

常见误区

  1. 忽略文档来源标注 — 生产环境必须标注来源，便于溯源和审核
  2. 不做相似度阈值检查 — 检索结果为空时应返回兜底回答
  3. 使用默认 temperature — RAG 场景应降低 temperature 减少幻觉
