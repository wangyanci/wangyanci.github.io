问题

如何防止 Agent 乱调用工具（工具调用幻觉）？有哪些工程上的防护策略？

答案

工具调用幻觉（Tool Calling Hallucination）是指 LLM 在不该调用工具时调用、选择不存在的工具、编造参数、漏填必填字段等问题。需要通过"定义层 + 校验层 + 执行层"的多层防线来防控。

工具调用幻觉的三种类型

类型 表现 根因
Type 1: 调错工具 选择了不相关或错误的工具 工具描述模糊、职责重叠
Type 2: 参数错误 编造参数、类型错误、漏填必填项 Schema 约束不足
Type 3: 无意义调用 闲聊场景也调工具、反复调同一工具 缺少调用必要性判断

五层防护体系

第一层：工具定义层（源头控制）

tool_schema = {
    "name": "query_order",
    "description": "查询订单信息。仅当用户提供了有效订单号时调用。"

第二层：调用前门禁层

class ToolCallGatekeeper:
    """工具调用前的门禁检查"""
    
    def __init__(self, registered_tools: set):
        self.registered_tools = registered_tools
        self.call_history = []
    
    async def check(self, tool_call: dict) -> tuple[bool, str]:
        """返回 (是否允许, 拒绝原因)"""
        
        # 1. 工具名白名单校验
        tool_name = tool_call.get("name")
        if tool_name not in self.registered_tools:
            return False, f"工具 '{tool_name}' 未注册"
        
        # 2. 重复调用检测
        recent_calls = [c for c in self.call_history[-5:] 
                       if c["name"] == tool_name and c["args"] == tool_call.get("arguments")]
        if len(recent_calls) >= 2:
            return False, "检测到重复调用，已拦截"
        
        # 3. 调用频率限制
        recent_same_tool = [c for c in self.call_history[-10:] 
                           if c["name"] == tool_name]
        if len(recent_same_tool) >= 5:
            return False, f"工具 '{tool_name}' 调用过于频繁"
        
        # 4. 高风险工具人工确认
        if tool_name in HIGH_RISK_TOOLS:
            return False, "高风险操作需要人工确认"
        
        self.call_history.append(tool_call)
        return True, ""

HIGH_RISK_TOOLS = {"delete_data", "transfer_money", "send_email", "modify_permissions"}

第三层：参数校验层

import jsonschema
from jsonschema import validate

class ParameterValidator:
    """参数严格校验"""
    
    def __init__(self, tool_schemas: dict):
        """tool_schemas: {tool_name: json_schema}"""
        self.schemas = tool_schemas
    
    def validate(self, tool_name: str, arguments: dict) -> tuple[bool, str]:
        schema = self.schemas.get(tool_name, {}).get("parameters", {})
        
        try:
            # 1. JSON Schema 校验
            validate(instance=arguments, schema=schema)
            
            # 2. 业务规则校验
            self._business_validation(tool_name, arguments)
            
            return True, ""
            
        except jsonschema.exceptions.ValidationError as e:
            return False, f"参数校验失败: {e.message}"
        except ValueError as e:
            return False, f"业务规则校验失败: {str(e)}"
    
    def _business_validation(self, tool_name: str, args: dict):
        """业务规则校验"""
        if tool_name == "query_order":
            order_id = args.get("order_id", "")
            if not order_id.startswith("ORD"):
                raise ValueError("订单号必须以 ORD 开头")
        
        elif tool_name == "date_range_query":
            start = args.get("start_date")
            end = args.get("end_date")
            if start and end and start > end:
                raise ValueError("开始日期不能晚于结束日期")
            # 限制查询范围，防止全表扫描
            if start and end and (end - start).days > 90:
                raise ValueError("查询时间范围不能超过 90 天")

第四层：执行安全层

class SecureToolExecutor:
    """安全的工具执行器"""
    
    def __init__(self, executor, audit_logger):
        self.executor = executor
        self.audit_logger = audit_logger
        self.approval_rules = {
            "read_only": ["query_*", "search_*", "get_*"],  # 自动执行
            "need_approval": ["delete_*", "update_*", "send_*"],  # 需审批
            "forbidden": ["drop_table", "exec_shell"]  # 禁止执行
        }
    
    async def execute(self, tool_call: dict, user_context: dict) -> dict:
        tool_name = tool_call["name"]
        
        # 1. 权限检查
        if not self._check_permission(tool_name, user_context):
            raise PermissionError(f"用户无权调用工具 '{tool_name}'")
        
        # 2. 审批检查
        risk_level = self._assess_risk(tool_name)
        if risk_level == "high" and not user_context.get("admin"):
            return {"status": "needs_approval", "tool": tool_name}
        
        # 3. 审计日志
        self.audit_logger.log({
            "event": "tool_call",
            "tool": tool_name,
            "args": tool_call.get("arguments"),
            "user": user_context.get("user_id"),
            "timestamp": time.time()
        })
        
        # 4. 超时控制
        try:
            result = await asyncio.wait_for(
                self.executor(tool_call),
                timeout=30
            )
            return {"status": "success", "result": result}
        except asyncio.TimeoutError:
            return {"status": "timeout", "error": "工具执行超时"}
    
    def _assess_risk(self, tool_name: str) -> str:
        for pattern in self.approval_rules["forbidden"]:
            if self._match(pattern, tool_name):
                return "forbidden"
        for pattern in self.approval_rules["need_approval"]:
            if self._match(pattern, tool_name):
                return "high"
        return "low"

第五层：观测反馈层

class ToolCallMonitor:
    """工具调用监控和反馈"""
    
    def __init__(self):
        self.call_records = []
    
    def record(self, tool_call: dict, result: dict, latency_ms: float):
        """记录调用信息"""
        self.call_records.append({
            "tool": tool_call["name"],
            "args": tool_call["arguments"],
            "result_status": result.get("status"),
            "latency_ms": latency_ms,
            "timestamp": time.time()
        })
    
    def get_metrics(self) -> dict:
        """计算监控指标"""
        total = len(self.call_records)
        if total == 0:
            return {}
        
        success = sum(1 for r in self.call_records if r["result_status"] == "success")
        failed = total - success
        
        return {
            "total_calls": total,
            "success_rate": success / total,
            "failure_rate": failed / total,
            "avg_latency_ms": sum(r["latency_ms"] for r in self.call_records) / total,
            "most_called": Counter(r["tool"] for r in self.call_records).most_common(5),
            "failure_reasons": Counter(
                r["result_status"] for r in self.call_records if r["result_status"] != "success"
            ).most_common()
        }

评估指标

指标 说明 目标值
工具选择准确率 选择了正确工具的比例 > 95%
参数有效率 参数完全符合 Schema 的比例 > 90%
无工具场景误调用率 不该调时却调用的比例 < 5%
越权拦截率 被权限系统拦截的调用比例 记录并分析

延伸追问

Q: 模型选择了正确工具但参数错了怎么办？
A: 返回结构化错误和缺失字段信息，让模型补充或向用户追问。不要用默认值偷偷执行高风险操作。

Q: 什么时候需要用户确认工具调用？
A: 涉及写操作、删除、付款、外发、权限变更、隐私数据、不可逆影响或高成本操作时需要确认。

常见误区

  1. 误区：只在执行层做校验就够了。
    纠正：必须从定义层就开始控制，五层防线缺一不可。

  2. 误区：工具调用失败就告诉用户"系统错误"。
    纠正：应该分类处理：可重试的重试，参数错误让模型修正，权限不足说明限制。

  3. 误区：所有工具走同样的安全策略。
    纠正：应该分级处理：只读查询自动执行，写操作需确认，高危操作需审批。
