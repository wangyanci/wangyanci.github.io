问题

在生产环境中，Agent 的工具调用系统面临哪些挑战？如何设计一个高扩展、安全、可观测的工具系统？请描述工具注册、发现、调用、监控的完整方案。

1. 答案

工具系统的核心挑战：

  1. 扩展性：工具数量可能从 10 个增长到 100+ 个
  2. 安全性：工具执行可能带来数据泄露、权限滥用等风险
  3. 可靠性：工具调用失败后的降级和重试
  4. 可观测性：谁调用了什么工具、参数是什么、结果如何
  5. 版本管理：工具接口变更的兼容性

高扩展性设计：

from abc import ABC, abstractmethod
from typing import Dict, List, Type, Optional
import asyncio

class BaseTool(ABC):
    """工具基类 - 所有工具必须实现"""
    
    name: str = ""
    description: str = ""
    version: str = "1.0"
    
    @property
    @abstractmethod
    def schema(self) -> dict:
        """返回 JSON Schema"""
        pass
    
    @abstractmethod
    async def run(self, **params) -> ToolResult:
        """执行工具"""
        pass
    
    @property
    def rate_limit(self) -> Optional[RateLimit]:
        """速率限制配置"""
        return None
    
    @property
    def required_permissions(self) -> List[str]:
        """所需权限列表"""
        return []


class ToolRegistry:
    """可扩展的工具注册中心"""
    
    def __init__(self):
        self._tools: Dict[str, BaseTool] = {}
        self._categories: Dict[str, List[str]] = {}
        self._middlewares: List[Middleware] = []
    
    def register(self, tool: BaseTool, category: str = "general"):
        """注册工具"""
        self._tools[tool.name] = tool
        
        if category not in self._categories:
            self._categories[category] = []
        self._categories[category].append(tool.name)
        
        print(f"工具注册成功: {tool.name} (v{tool.version})")
    
    def get_tool(self, name: str) -> Optional[BaseTool]:
        return self._tools.get(name)
    
    def get_schemas_by_category(self, category: str) -> List[dict]:
        """按类别获取工具 Schema（支持动态加载）"""
        tool_names = self._categories.get(category, [])
        return [self._tools[name].schema for name in tool_names 
                if name in self._tools]
    
    def list_tools(self) -> List[dict]:
        """列出所有工具"""
        return [
            {"name": name, "description": tool.description, 
             "version": tool.version, "category": cat}
            for cat, names in self._categories.items()
            for name in names
            if (tool := self._tools.get(name))
        ]


# 动态工具加载（插件化）
class PluginLoader:
    """插件化工具加载器"""
    
    def __init__(self, registry: ToolRegistry):
        self.registry = registry
        self.plugins: Dict[str, Plugin] = {}
    
    async def load_plugin(self, plugin_path: str):
        """动态加载工具插件"""
        module = importlib.import_module(plugin_path)
        
        for name, obj in inspect.getmembers(module):
            if (inspect.isclass(obj) and 
                issubclass(obj, BaseTool) and 
                obj != BaseTool):
                
                tool = obj()
                self.registry.register(tool, category=plugin_path)
                self.plugins[name] = plugin_path

安全性设计：

class SecureToolExecutor:
    """安全的工具执行器"""
    
    def __init__(self):
        self.sandbox = DockerSandbox()  # 沙箱执行环境
        self.audit_logger = AuditLogger()
        self.policy_engine = PolicyEngine()
    
    async def execute(self, tool: BaseTool, params: dict, 
                      context: ExecutionContext) -> ToolResult:
        """带安全控制的工具执行"""
        
        # 1. 权限检查
        for permission in tool.required_permissions:
            if not context.user.has_permission(permission):
                self.audit_logger.log_access_denied(context, tool, permission)
                raise PermissionError(f"缺少权限: {permission}")
        
        # 2. 输入校验
        validated = self.validate_params(tool.schema, params)
        if not validated.is_valid:
            return ToolResult.error(f"参数校验失败: {validated.errors}")
        
        # 3. 策略检查
        policy_check = await self.policy_engine.check(
            tool=tool, params=params, context=context
        )
        if not policy_check.allowed:
            self.audit_logger.log_policy_violation(context, tool, policy_check)
            return ToolResult.error(f"策略阻止: {policy_check.reason}")
        
        # 4. 速率限制
        if tool.rate_limit:
            allowed = await self.check_rate_limit(tool, context.user)
            if not allowed:
                return ToolResult.error("速率限制 exceeded")
        
        # 5. 沙箱执行
        with self.audit_logger.record_execution(context, tool, params):
            try:
                if tool.is_dangerous:
                    result = await self.sandbox.run(tool, validated.params)
                else:
                    result = await tool.run(**validated.params)
                
                return ToolResult.success(result)
                
            except Exception as e:
                self.audit_logger.log_error(context, tool, e)
                return ToolResult.error(str(e))
    
    def validate_params(self, schema: dict, params: dict) -> ValidationResult:
        """JSON Schema 参数校验"""
        try:
            jsonschema.validate(params, schema)
            return ValidationResult(valid=True)
        except jsonschema.ValidationError as e:
            return ValidationResult(valid=False, errors=str(e))

可观测性设计：

class ToolMetrics:
    """工具调用指标监控"""
    
    def __init__(self):
        self.call_count = Counter("tool_calls_total", ["tool_name", "status"])
        self.latency = Histogram("tool_latency_seconds", ["tool_name"])
        self.errors = Counter("tool_errors_total", ["tool_name", "error_type"])
        self.active_calls = Gauge("tool_active_calls", ["tool_name"])
    
    def record_call(self, tool_name: str, latency: float, success: bool):
        status = "success" if success else "error"
        self.call_count.labels(tool_name=tool_name, status=status).inc()
        self.latency.labels(tool_name=tool_name).observe(latency)


class AuditLogger:
    """审计日志"""
    
    def record_execution(self, context, tool, params):
        log_entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "user_id": context.user.id,
            "session_id": context.session_id,
            "tool_name": tool.name,
            "tool_version": tool.version,
            "params": self.sanitize_params(params),  # 脱敏
            "ip_address": context.ip_address,
        }
        
        # 写入审计日志（不可篡改存储）
        self.audit_store.append(log_entry)
        
        return AuditContext(log_entry)
    
    def sanitize_params(self, params: dict) -> dict:
        """敏感参数脱敏"""
        sanitized = {}
        for k, v in params.items():
            if any(s in k.lower() for s in ["password", "token", "secret", "key"]):
                sanitized[k] = "***REDACTED***"
            else:
                sanitized[k] = v
        return sanitized

2. 延伸追问

  1. 追问：工具数量很多时（100+），LLM 如何选择工具？

    • 答：①工具分层（只注入相关类别的工具）②工具描述向量化，先做相似度筛选 ③使用 tool retrieval 机制

  2. 追问：如何防止 Prompt 注入攻击通过工具参数？

    • 答：①严格的输入校验 ②参数类型和范围限制 ③敏感字符过滤 ④工具执行沙箱化

3. 常见误区

  • 误区：工具越多越好
  • 纠正：工具过多会降低 LLM 选择准确性。建议每次只暴露 ≤8 个最相关的工具
