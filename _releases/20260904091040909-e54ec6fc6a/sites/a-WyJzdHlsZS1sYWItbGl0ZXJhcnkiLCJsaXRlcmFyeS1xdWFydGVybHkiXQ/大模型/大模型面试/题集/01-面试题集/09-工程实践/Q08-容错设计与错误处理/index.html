Agent 的容错设计和错误处理机制如何设计？

问题

Agent 在执行复杂任务时会遇到各种错误，如何设计健壮的容错和错误恢复机制？

答案

一、分层错误分类与处理

class ErrorClassifier:
    """错误分类器"""
    
    ERROR_TYPES = {
        # === 可恢复错误 (Retryable) ===
        "TRANSIENT": {
            "examples": ["网络超时", "API限流", "临时服务不可用"],
            "strategy": "指数退避重试（最多3次）",
            "retry": True
        },
        
        # === 可降级错误 (Degradable) ===
        "DEGRADABLE": {
            "examples": ["主模型不可用", "工具服务故障"],
            "strategy": "切换到备用模型/工具",
            "fallback": True
        },
        
        # === 用户可修正错误 (UserFixable) ===
        "USER_FIXABLE": {
            "examples": ["参数不完整", "权限不足", "输入格式错误"],
            "strategy": "向用户请求澄清",
            "clarify": True
        },
        
        # === 不可恢复错误 (Fatal) ===
        "FATAL": {
            "examples": ["数据损坏", "配置错误", "系统性故障"],
            "strategy": "记录日志，转人工处理",
            "escalate": True
        }
    }

二、Agent 级别自愈设计

class SelfHealingAgent:
    """具备自愈能力的 Agent"""
    
    async def execute_with_recovery(self, task: Task) -> Result:
        state = TaskState(task)
        
        for step in range(self.MAX_STEPS):
            try:
                action = await self._plan_next_action(state)
                observation = await self._execute_action(action)
                state.add_observation(observation)
                
                if state.is_complete():
                    return state.get_result()
                    
            except ToolExecutionError as e:
                # 工具调用失败时的自愈逻辑
                recovery = await self._decide_recovery(state, e)
                
                if recovery == "retry":
                    continue  # 重试当前步骤
                elif recovery == "alternative_tool":
                    state.use_alternative_tool()  # 使用替代工具
                elif recovery == "skip":
                    state.mark_skipped()  # 跳过此步骤
                elif recovery == "ask_user":
                    return await self._ask_user_for_help(state, e)
                else:
                    return Result.error(f"无法恢复: {e}")
        
        return Result.error("超过最大步数限制")
    
    async def _decide_recovery(self, state: TaskState, error: Exception) -> str:
        """让 LLM 决定恢复策略"""
        prompt = f"""
        当前任务状态：{state.summary()}
        遇到错误：{error}
        可用恢复选项：retry, alternative_tool, skip, ask_user
        请选择最佳恢复策略并说明原因：
        """
        decision = await self.llm.invoke(prompt)
        return decision.content.strip()

三、断点续传机制

class CheckpointManager:
    """断点续传管理器"""
    
    def __init__(self, redis_client: Redis):
        self.redis = redis_client
        self.checkpoint_interval = 3  # 每 3 步保存一次
    
    async def save_checkpoint(self, session_id: str, state: AgentState):
        """保存检查点"""
        checkpoint = {
            "timestamp": time.time(),
            "current_step": state.current_step,
            "memory": state.memory.serialize(),
            "intermediate_results": state.intermediate_results,
            "tool_call_history": state.tool_calls
        }
        await self.redis.setex(
            f"checkpoint:{session_id}",
            ttl=86400,  # 24h 过期
            value=json.dumps(checkpoint)
        )
    
    async def restore_checkpoint(self, session_id: str) -> AgentState | None:
        """从检查点恢复"""
        data = await self.redis.get(f"checkpoint:{session_id}")
        if data:
            checkpoint = json.loads(data)
            return AgentState.deserialize(checkpoint)
        return None
