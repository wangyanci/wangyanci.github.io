问题

Function Call 为什么会产生工具调用幻觉？如何从工程上系统化地降低误调用率？

答案

工具调用幻觉不仅是模型"随机犯错"，更是系统设计缺陷的表现。需要从根因分析出发，建立"定义-门禁-校验-反馈-观测"的全链路治理体系。

# 工具职责互斥性检查
def check_tool_separation(tools: list) -> list:
    """检查工具间是否有职责重叠"""
    issues = []
    for i, t1 in enumerate(tools):
        for t2 in tools[i+1:]:
            # 计算描述语义相似度
            sim = semantic_similarity(t1["description"], t2["description"])
            if sim > 0.6:
                issues.append({
                    "tools": [t1["name"], t2["name"]],
                    "similarity": sim,
                    "suggestion": f"请明确区分 {t1['name']} 和 {t2['name']} 的适用边界"
                })
    return issues

第二层：调用门禁层（拦截不合理调用）

class ToolCallGate:
    """调用前的多层门禁"""
    
    async def evaluate(self, context: dict, tool_call: dict) -> dict:
        checks = [
            ("necessity", self._check_necessity),      # 必要性检查
            ("permission", self._check_permission),     # 权限检查
            ("parameter", self._check_parameters),      # 参数检查
            ("frequency", self._check_frequency),       # 频率检查
            ("risk", self._check_risk)                  # 风险检查
        ]
        
        results = {}
        for check_name, check_func in checks:
            result = await check_func(context, tool_call)
            results[check_name] = result
            if not result["pass"]:
                return {
                    "allowed": False,
                    "reason": result["reason"],
                    "failed_check": check_name
                }
        
        return {"allowed": True, "checks": results}
    
    async def _check_necessity(self, context: dict, tool_call: dict) -> dict:
        """检查调用是否必要"""
        user_message = context.get("last_user_message", "")
        
        # 简单规则：闲聊类消息不需要调工具
        chat_patterns = ["你好", "谢谢", "再见", "哈哈", "嗯"]
        if any(p in user_message for p in chat_patterns) and len(user_message) < 10:
            return {"pass": False, "reason": "闲聊场景不需要调用工具"}
        
        return {"pass": True}
    
    async def _check_frequency(self, context: dict, tool_call: dict) -> dict:
        """检查调用频率"""
        tool_name = tool_call["name"]
        recent_calls = context.get("recent_tool_calls", [])
        
        # 同一工具 1 分钟内调用不超过 3 次
        same_tool_recent = [
            c for c in recent_calls 
            if c["name"] == tool_name and c["time"] > time.time() - 60
        ]
        if len(same_tool_recent) >= 3:
            return {"pass": False, "reason": f"工具 {tool_name} 调用过于频繁"}
        
        # 总调用次数限制
        if len(recent_calls) > 10:
            return {"pass": False, "reason": "单会话工具调用次数超限"}
        
        return {"pass": True}

第三层：参数校验层（结构化约束）

class StrictParameterValidator:
    """严格的参数校验器"""
    
    async def validate(self, schema: dict, arguments: dict) -> dict:
        errors = []
        
        # 1. JSON Schema 校验
        from jsonschema import validate, ValidationError
        try:
            validate(arguments, schema)
        except ValidationError as e:
            errors.append(f"Schema校验失败: {e.message}")
        
        # 2. 必填字段校验（即使 Schema 已通过，再次确认）
        required = schema.get("required", [])
        for field in required:
            if field not in arguments or arguments[field] is None:
                errors.append(f"必填字段缺失: {field}")
        
        # 3. 业务规则校验
        errors.extend(self._business_rules(schema, arguments))
        
        # 4. 安全校验
        errors.extend(self._security_rules(schema, arguments))
        
        if errors:
            return {
                "valid": False,
                "errors": errors,
                "suggestion": self._generate_fix_suggestion(errors, schema)
            }
        return {"valid": True}
    
    def _security_rules(self, schema: dict, arguments: dict) -> list:
        """安全规则校验"""
        errors = []
        
        # 检查路径穿越
        for key, value in arguments.items():
            if isinstance(value, str):
                if ".." in value or value.startswith("/"):
                    if "path" in key.lower() or "file" in key.lower():
                        errors.append(f"字段 {key} 包含非法路径: {value}")
                
                # 检查 SQL 注入
                if "sql" in key.lower() or "query" in key.lower():
                    dangerous = ['drop', 'delete', 'update', 'insert', '--', ';--']
                    if any(d in value.lower() for d in dangerous):
                        errors.append(f"字段 {key} 可能包含 SQL 注入")
        
        return errors

第四层：执行反馈层（结构化错误处理）

class StructuredErrorFeedback:
    """结构化的错误反馈"""
    
    def format_error(self, error: Exception, tool_name: str) -> dict:
        """将错误转换为模型可理解的结构化信息"""
        
        error_types = {
            TimeoutError: {
                "type": "timeout",
                "retryable": True,
                "suggestion": "服务暂时不可用，请稍后重试或简化请求"
            },
            ConnectionError: {
                "type": "connection_failed",
                "retryable": True,
                "suggestion": "网络连接失败，请检查网络后重试"
            },
            PermissionError: {
                "type": "permission_denied",
                "retryable": False,
                "suggestion": "权限不足，请联系管理员授权"
            },
            ValueError: {
                "type": "invalid_parameter",
                "retryable": True,
                "suggestion": "参数校验失败，请检查参数格式和范围"
            }
        }
        
        error_info = error_types.get(type(error), {
            "type": "unknown",
            "retryable": False,
            "suggestion": "未知错误，请联系技术支持"
        })
        
        return {
            "tool": tool_name,
            "status": "failed",
            "error": {
                "type": error_info["type"],
                "message": str(error),
                "retryable": error_info["retryable"]
            },
            "suggestion": error_info["suggestion"]
        }

第五层：观测评测层（持续优化）

class ToolCallObservability:
    """工具调用可观测性系统"""
    
    def __init__(self):
        self.metrics = {
            "total_calls": 0,
            "success_calls": 0,
            "failed_calls": 0,
            "hallucination_cases": [],
            "parameter_errors": [],
            "latency_ms": []
        }
    
    def record(self, tool_call: dict, result: dict, latency_ms: float):
        self.metrics["total_calls"] += 1
        self.metrics["latency_ms"].append(latency_ms)
        
        if result.get("status") == "success":
            self.metrics["success_calls"] += 1
        else:
            self.metrics["failed_calls"] += 1
            
            # 分类记录失败原因
            error_type = result.get("error", {}).get("type", "unknown")
            if error_type in ["tool_not_found", "invalid_parameter"]:
                self.metrics["hallucination_cases"].append({
                    "tool_call": tool_call,
                    "error": result,
                    "timestamp": time.time()
                })
    
    def get_hallucination_rate(self) -> float:
        total = self.metrics["total_calls"]
        if total == 0:
            return 0.0
        hallucination_count = len(self.metrics["hallucination_cases"])
        return hallucination_count / total
    
    def generate_report(self) -> dict:
        return {
            "summary": {
                "total_calls": self.metrics["total_calls"],
                "success_rate": self.metrics["success_calls"] / max(self.metrics["total_calls"], 1),
                "avg_latency_ms": sum(self.metrics["latency_ms"]) / max(len(self.metrics["latency_ms"]), 1),
                "hallucination_rate": self.get_hallucination_rate()
            },
            "top_hallucination_tools": self._get_top_hallucination_tools(),
            "improvement_suggestions": self._generate_suggestions()
        }
