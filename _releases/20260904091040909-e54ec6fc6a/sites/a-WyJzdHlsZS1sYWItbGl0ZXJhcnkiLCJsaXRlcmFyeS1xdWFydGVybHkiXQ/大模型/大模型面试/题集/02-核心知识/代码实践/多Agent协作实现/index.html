3.2. 工具定义

"""
tools/search_tools.py
自定义搜索工具
"""

from crewai_tools import BaseTool
from typing import Type, Optional
from pydantic import BaseModel, Field
import requests


class SearchInput(BaseModel):
    """搜索工具输入参数"""
    query: str = Field(..., description="搜索查询")
    max_results: int = Field(default=5, description="最大结果数")


class WebSearchTool(BaseTool):
    """网页搜索工具"""
    
    name: str = "web_search"
    description: str = "使用搜索引擎查询信息"
    args_schema: Type[BaseModel] = SearchInput
    
    def _run(self, query: str, max_results: int = 5) -> str:
        """
        执行搜索
        
        实际项目中应使用 SerpAPI、Bing API 等
        这里模拟搜索结果
        """
        # 模拟搜索结果
        mock_results = {
            "Python": "Python 是一种高级编程语言，由 Guido van Rossum 于 1991 年创建。",
            "机器学习": "机器学习是 AI 的分支，通过数据训练模型进行预测和决策。",
            "深度学习": "深度学习使用多层神经网络，在图像识别、NLP 等领域表现优异。",
        }
        
        results = []
        for key, value in mock_results.items():
            if query.lower() in key.lower() or query.lower() in value.lower():
                results.append(f"[{key}] {value}")
        
        if not results:
            return f"未找到关于 '{query}' 的直接结果。请尝试其他关键词。"
        
        return "\n\n".join(results[:max_results])


class DocumentSearchTool(BaseTool):
    """文档搜索工具"""
    
    name: str = "document_search"
    description: str = "在内部文档中搜索信息"
    args_schema: Type[BaseModel] = SearchInput
    
    # 模拟内部知识库
    KNOWLEDGE_BASE = {
        "公司介绍": "我们是一家专注于 AI 技术的科技公司，成立于 2020 年。",
        "产品信息": "我们的主要产品包括 AI Agent 平台和智能客服系统。",
        "技术栈": "我们使用 Python、LangChain、CrewAI 等技术构建产品。",
    }
    
    def _run(self, query: str, max_results: int = 5) -> str:
        """在内部文档中搜索"""
        results = []
        for key, value in self.KNOWLEDGE_BASE.items():
            if query.lower() in key.lower() or query.lower() in value.lower():
                results.append(f"[{key}] {value}")
        
        if not results:
            return f"内部文档中未找到 '{query}' 相关信息。"
        
        return "\n\n".join(results[:max_results])

"""

from crewai import Agent
from tools.search_tools import WebSearchTool, DocumentSearchTool


def create_researcher() -> Agent:
    """
    创建研究员 Agent
    
    负责信息收集和调研
    """
    return Agent(
        role="高级研究员",
        goal="深入调研指定主题，收集全面、准确的信息",
        backstory="""你是一位经验丰富的研究员，擅长信息检索和分析。
你做事严谨，注重细节，能够从多个来源收集信息并进行综合分析。
你总是确保信息的准确性和时效性。""",
        verbose=True,
        allow_delegation=False,
        tools=[WebSearchTool(), DocumentSearchTool()]
    )

"""
tasks/research_task.py
研究任务

from crewai import Task
from agents.researcher import create_researcher


def create_research_task(topic: str) -> Task:
    """
    创建研究任务
    
    Args:
        topic: 研究主题
    """
    researcher = create_researcher()
    
    return Task(
        description=f"""对以下主题进行深入研究：{topic}

要求：
1. 收集该主题的核心概念和关键信息
2. 查找相关的技术细节和最新进展
3. 整理主要观点和关键数据
4. 识别该主题的重要性和应用场景

请提供一份结构化的研究报告，包含：
- 主题概述
- 核心概念解释
- 关键数据和事实
- 最新进展
- 应用场景
""",
        expected_output="一份详细、结构化的研究报告，包含主题的所有关键信息",
        agent=researcher
    )

"""
tasks/write_task.py
写作任务

from crewai import Task

def create_write_task(topic: str) -> Task:

    创建写作任务

1. 标题吸引人且准确反映内容

文章格式：
# 标题

## 引言
...

## 主体内容
...

## 总结与展望
...
""",
        expected_output="一篇结构完整、内容丰富的文章，Markdown 格式",
        agent=writer,
        context=[]  # 将在运行时填充研究报告

3.5. Crew 编排

"""
crew.py
Crew 编排
"""

from crewai import Crew, Process
from agents.researcher import create_researcher
from agents.writer import create_writer
from agents.reviewer import create_reviewer
from tasks.research_task import create_research_task
from tasks.write_task import create_write_task
from tasks.review_task import create_review_task


class ContentCreationCrew:
    """
    内容创作团队
    
    研究员 + 写作专家 + 审校专家 协作完成内容创作
    """
    
    def __init__(self, topic: str, process_type: str = "sequential"):
        self.topic = topic
        self.process_type = process_type
        
        # 创建 Agent
        self.researcher = create_researcher()
        self.writer = create_writer()
        self.reviewer = create_reviewer()
        
        # 创建任务
        self.research_task = create_research_task(topic)
        self.write_task = create_write_task(topic)
        self.review_task = create_review_task(topic)
        
        # 设置任务依赖
        self.write_task.context = [self.research_task]
        self.review_task.context = [self.write_task]
    
    def run(self) -> str:
        """
        运行 Crew
        
        Returns:
            最终结果
        """
        # 选择流程类型
        process = Process.sequential
        if self.process_type == "hierarchical":
            process = Process.hierarchical
        
        # 创建 Crew
        crew = Crew(
            agents=[self.researcher, self.writer, self.reviewer],
            tasks=[self.research_task, self.write_task, self.review_task],
            process=process,
            verbose=True
        )
        
        # 执行
        result = crew.kickoff()
        
        return result


class ParallelResearchCrew:
    """
    并行研究团队
    
    多个研究员同时从不同角度研究，然后汇总
    """
    
    def __init__(self, topic: str):
        self.topic = topic
    
    def run(self) -> str:
        """并行执行研究"""
        # 创建多个研究员（不同角度）
        tech_researcher = Agent(
            role="技术研究专家",
            goal=f"从技术角度深入研究 {self.topic}",
            backstory="你专注于技术实现和架构分析",
            verbose=True
        )
        
        market_researcher = Agent(
            role="市场研究专家",
            goal=f"从市场角度研究 {self.topic}",
            backstory="你专注于市场趋势和商业应用",
            verbose=True
        )
        
        # 创建任务
        tech_task = Task(
            description=f"从技术角度研究 {self.topic}，关注技术原理和实现",
            expected_output="技术分析报告",
            agent=tech_researcher
        )
        
        market_task = Task(
            description=f"从市场角度研究 {self.topic}，关注应用场景和商业价值",
            expected_output="市场分析报告",
            agent=market_researcher
        )
        
        # 汇总 Agent
        synthesizer = Agent(
            role="研究汇总专家",
            goal="整合多个研究报告",
            backstory="你擅长整合多来源信息",
            verbose=True
        )
        
        synthesize_task = Task(
            description="整合技术和市场研究报告，形成综合报告",
            expected_output="综合研究报告",
            agent=synthesizer,
            context=[tech_task, market_task]
        )
        
        # 并行执行
        crew = Crew(
            agents=[tech_researcher, market_researcher, synthesizer],
            tasks=[tech_task, market_task, synthesize_task],
            process=Process.parallel,
            verbose=True
        )
        
        return crew.kickoff()

3.6. 主程序

"""
main.py
入口文件
"""

from crew import ContentCreationCrew, ParallelResearchCrew


def main():
    """主程序"""
    print("🤖 多 Agent 协作系统")
    print("=" * 60)
    
    # 获取主题
    topic = input("请输入研究主题: ").strip()
    
    if not topic:
        topic = "AI Agent 技术"
    
    print(f"\n📋 主题: {topic}")
    print("请选择模式:")
    print("1. 顺序模式 (研究 → 写作 → 审校)")
    print("2. 层级模式 (Manager 协调)")
    print("3. 并行研究 (多角度同时研究)")
    
    choice = input("请选择 (1/2/3): ").strip()
    
    if choice == "3":
        print("\n🚀 启动并行研究...")
        crew = ParallelResearchCrew(topic)
    else:
        process = "sequential" if choice == "1" else "hierarchical"
        print(f"\n🚀 启动 {process} 模式...")
        crew = ContentCreationCrew(topic, process_type=process)
    
    # 运行
    result = crew.run()
    
    # 输出结果
    print("\n" + "=" * 60)
    print("✅ 任务完成!")
    print("=" * 60)
    print(result)


if __name__ == "__main__":
    main()
