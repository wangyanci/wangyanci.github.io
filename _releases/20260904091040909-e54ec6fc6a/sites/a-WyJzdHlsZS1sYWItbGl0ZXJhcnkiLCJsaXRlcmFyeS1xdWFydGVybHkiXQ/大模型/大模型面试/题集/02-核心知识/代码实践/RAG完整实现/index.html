3.1 基础配置

"""
RAG 完整实现
基于 LangChain 的检索增强生成系统
"""

import os
from typing import List, Dict, Optional
from dataclasses import dataclass

from langchain_core.documents import Document
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough
from langchain_core.output_parsers import StrOutputParser

from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_chroma import Chroma

from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.document_loaders import (
    TextLoader,
    DirectoryLoader,
    PyPDFLoader
)


# ============== 配置 ==============

@dataclass
class RAGConfig:
    """RAG 配置"""
    # OpenAI 配置
    openai_api_key: str = ""
    model_name: str = "gpt-4"
    embedding_model: str = "text-embedding-3-small"
    
    # 文档处理
    chunk_size: int = 500
    chunk_overlap: int = 50
    
    # 检索配置
    top_k: int = 5
    score_threshold: float = 0.5
    
    # 向量数据库
    persist_directory: str = "./chroma_db"
    collection_name: str = "rag_collection"


# 全局配置
config = RAGConfig(
    openai_api_key=os.getenv("OPENAI_API_KEY", "")
)

3.2 文档处理模块

# ============== 文档处理 ==============

class DocumentProcessor:
    """文档处理器"""
    
    def __init__(self, cfg: RAGConfig = None):
        self.cfg = cfg or config
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=self.cfg.chunk_size,
            chunk_overlap=self.cfg.chunk_overlap,
            length_function=len,
            separators=["\n\n", "\n", "。", " ", ""]
        )
    
    def load_file(self, file_path: str) -> List[Document]:
        """
        加载单个文件
        
        支持格式: .txt, .md, .pdf
        """
        if not os.path.exists(file_path):
            raise FileNotFoundError(f"文件不存在: {file_path}")
        
        # 根据扩展名选择加载器
        if file_path.endswith('.pdf'):
            loader = PyPDFLoader(file_path)
        else:
            loader = TextLoader(file_path, encoding='utf-8')
        
        documents = loader.load()
        print(f"📄 加载文件: {file_path}, 共 {len(documents)} 页/段")
        return documents
    
    def load_directory(self, dir_path: str, glob: str = "**/*.txt") -> List[Document]:
        """
        加载目录下的所有文件
        
        Args:
            dir_path: 目录路径
            glob: 文件匹配模式
        """
        loader = DirectoryLoader(
            dir_path,
            glob=glob,
            loader_cls=TextLoader,
            loader_kwargs={'encoding': 'utf-8'}
        )
        documents = loader.load()
        print(f"📁 加载目录: {dir_path}, 共 {len(documents)} 个文件")
        return documents
    
    def split_documents(self, documents: List[Document]) -> List[Document]:
        """
        分块处理
        
        使用递归字符分割器，优先按段落分割
        """
        chunks = self.text_splitter.split_documents(documents)
        print(f"✂️ 分块完成: {len(documents)} 个文档 → {len(chunks)} 个块")
        
        # 显示分块统计
        lengths = [len(chunk.page_content) for chunk in chunks]
        print(f"   块大小统计: min={min(lengths)}, max={max(lengths)}, "
              f"avg={sum(lengths)//len(lengths)}")
        
        return chunks
    
    def add_metadata(self, chunks: List[Document], metadata: Dict = None) -> List[Document]:
        """
        添加元数据到文档块
        
        可用于后续过滤检索
        """
        for i, chunk in enumerate(chunks):
            chunk.metadata.update({
                'chunk_index': i,
                'chunk_size': len(chunk.page_content),
                'processing_time': __import__('time').time(),
                **(metadata or {})
            })
        return chunks

3.3 向量存储模块

# ============== 向量存储 ==============

class VectorStore:
    """向量存储管理"""
    
    def __init__(self, cfg: RAGConfig = None):
        self.cfg = cfg or config
        self.embeddings = OpenAIEmbeddings(
            api_key=self.cfg.openai_api_key,
            model=self.cfg.embedding_model
        )
        self.vectorstore: Optional[Chroma] = None
    
    def create_index(self, documents: List[Document]) -> Chroma:
        """
        从文档创建向量索引
        
        Args:
            documents: 文档块列表
            
        Returns:
            Chroma 向量数据库实例
        """
        print(f"🔧 创建向量索引...")
        print(f"   Embedding 模型: {self.cfg.embedding_model}")
        
        self.vectorstore = Chroma.from_documents(
            documents=documents,
            embedding=self.embeddings,
            persist_directory=self.cfg.persist_directory,
            collection_name=self.cfg.collection_name
        )
        
        print(f"✅ 索引创建完成")
        print(f"   存储路径: {self.cfg.persist_directory}")
        
        return self.vectorstore
    
    def load_index(self) -> Chroma:
        """
        加载已存在的向量索引
        """
        if not os.path.exists(self.cfg.persist_directory):
            raise FileNotFoundError(f"索引不存在: {self.cfg.persist_directory}")
        
        self.vectorstore = Chroma(
            persist_directory=self.cfg.persist_directory,
            embedding_function=self.embeddings,
            collection_name=self.cfg.collection_name
        )
        
        print(f"📂 加载已有索引")
        return self.vectorstore
    
    def add_documents(self, documents: List[Document]):
        """
        向已有索引添加文档
        """
        if self.vectorstore is None:
            raise ValueError("请先创建或加载索引")
        
        self.vectorstore.add_documents(documents)
        print(f"➕ 添加 {len(documents)} 个文档块到索引")
    
    def similarity_search(
        self, 
        query: str, 
        top_k: int = None,
        filter_dict: Dict = None
    ) -> List[Document]:
        """
        相似度检索
        
        Args:
            query: 查询文本
            top_k: 返回数量
            filter_dict: 元数据过滤条件
            
        Returns:
            相关文档列表
        """
        if self.vectorstore is None:
            raise ValueError("请先创建或加载索引")
        
        k = top_k or self.cfg.top_k
        
        results = self.vectorstore.similarity_search(
            query=query,
            k=k,
            filter=filter_dict
        )
        
        return results
    
    def similarity_search_with_score(
        self, 
        query: str, 
        top_k: int = None
    ) -> List[tuple]:
        """
        带分数的相似度检索
        
        Returns:
            [(Document, score), ...]
        """
        if self.vectorstore is None:
            raise ValueError("请先创建或加载索引")
        
        k = top_k or self.cfg.top_k
        
        results = self.vectorstore.similarity_search_with_score(
            query=query,
            k=k
        )
        
        return results

3.4 RAG 链构建

# ============== RAG 链 ==============

class RAGChain:
    """
    RAG 完整链
    
    封装检索 + 生成的完整流程
    """
    
    # RAG Prompt 模板
    RAG_TEMPLATE = """你是一个专业的 AI 助手。请基于以下检索到的上下文信息回答用户问题。

## 重要规则
1. 只使用提供的上下文信息回答问题
2. 如果上下文中没有足够信息，请明确说明
3. 回答应简洁、准确
4. 如涉及数据，请保持精确

## 上下文
{context}

## 用户问题
{question}

请回答："""
    
    def __init__(
        self, 
        vectorstore: VectorStore,
        cfg: RAGConfig = None
    ):
        self.cfg = cfg or config
        self.vectorstore = vectorstore
        self.llm = ChatOpenAI(
            api_key=self.cfg.openai_api_key,
            model=self.cfg.model_name,
            temperature=0.7
        )
        self.prompt = ChatPromptTemplate.from_template(self.RAG_TEMPLATE)
    
    def _format_context(self, documents: List[Document]) -> str:
        """
        格式化检索结果作为上下文
        
        添加来源标注，便于追溯
        """
        context_parts = []
        for i, doc in enumerate(documents, 1):
            source = doc.metadata.get('source', '未知来源')
            content = doc.page_content.strip()
            context_parts.append(
                f"[文档 {i}] 来源: {source}\n{content}"
            )
        
        return "\n\n---\n\n".join(context_parts)
    
    def _retrieve(self, query: str) -> List[Document]:
        """检索相关文档"""
        results = self.vectorstore.similarity_search_with_score(query)
        
        # 过滤低分结果
        filtered = [
            doc for doc, score in results 
            if score >= self.cfg.score_threshold
        ]
        
        return filtered or [doc for doc, _ in results[:3]]
    
    def build_chain(self):
        """
        构建 LCEL 链
        
        使用 LangChain Expression Language 声明式构建
        """
        chain = (
            {
                "context": lambda x: self._format_context(
                    self._retrieve(x["question"])
                ),
                "question": RunnablePassthrough()
            }
            | self.prompt
            | self.llm
            | StrOutputParser()
        )
        
        return chain
    
    def query(self, question: str) -> Dict:
        """
        执行 RAG 查询
        
        Args:
            question: 用户问题
            
        Returns:
            包含回答、上下文、来源的字典
        """
        # 检索
        retrieved_docs = self._retrieve(question)
        context = self._format_context(retrieved_docs)
        
        # 构建输入
        chain_input = {
            "context": context,
            "question": question
        }
        
        # 生成
        messages = self.prompt.format_messages(**chain_input)
        response = self.llm.invoke(messages)
        answer = response.content
        
        # 收集来源
        sources = [
            {
                "source": doc.metadata.get('source', '未知'),
                "content": doc.page_content[:200] + "..."
            }
            for doc in retrieved_docs
        ]
        
        return {
            "question": question,
            "answer": answer,
            "context": context,
            "sources": sources,
            "retrieved_count": len(retrieved_docs)
        }

3.5 RAG 系统封装

# ============== RAG 系统 ==============

class RAGSystem:
    """
    RAG 系统封装
    
    提供高级接口，简化使用
    """
    
    def __init__(self, cfg: RAGConfig = None):
        self.cfg = cfg or config
        self.processor = DocumentProcessor(self.cfg)
        self.vectorstore = VectorStore(self.cfg)
        self.rag_chain: Optional[RAGChain] = None
    
    def ingest(self, source: str, is_directory: bool = False):
        """
        摄入文档
        
        Args:
            source: 文件或目录路径
            is_directory: 是否为目录
        """
        print(f"\n{'='*50}")
        print(f"📥 开始摄入文档: {source}")
        print(f"{'='*50}")
        
        # 1. 加载文档
        if is_directory:
            documents = self.processor.load_directory(source)
        else:
            documents = self.processor.load_file(source)
        
        # 2. 分块
        chunks = self.processor.split_documents(documents)
        
        # 3. 添加元数据
        chunks = self.processor.add_metadata(chunks)
        
        # 4. 创建索引
        self.vectorstore.create_index(chunks)
        
        # 5. 初始化 RAG 链
        self.rag_chain = RAGChain(self.vectorstore, self.cfg)
        
        print(f"✅ 文档摄入完成!\n")
    
    def ask(self, question: str) -> str:
        """
        问答接口
        
        Args:
            question: 用户问题
            
        Returns:
            回答文本
        """
        if self.rag_chain is None:
            raise ValueError("请先摄入文档 (调用 ingest 方法)")
        
        print(f"\n❓ 问题: {question}")
        print("-" * 50)
        
        result = self.rag_chain.query(question)
        
        print(f"✅ 回答:\n{result['answer']}\n")
        print(f"📚 参考来源 ({result['retrieved_count']} 个):")
        for i, source in enumerate(result['sources'], 1):
            print(f"   {i}. {source['source']}")
        
        return result['answer']
    
    def chat(self):
        """
        交互式对话模式
        """
        if self.rag_chain is None:
            raise ValueError("请先摄入文档 (调用 ingest 方法)")
        
        print("\n" + "="*50)
        print("🤖 RAG 对话模式 (输入 'quit' 退出)")
        print("="*50 + "\n")
        
        while True:
            question = input("你: ").strip()
            
            if question.lower() in ['quit', 'exit', 'q']:
                print("再见!")
                break
            
            if not question:
                continue
            
            try:
                self.ask(question)
            except Exception as e:
                print(f"❌ 错误: {e}")

混合检索（向量 + 关键词）

from langchain.retrievers import EnsembleRetriever
from langchain_community.retrievers import BM25Retriever

def create_hybrid_retriever(documents: List[Document], vectorstore):
    """创建混合检索器"""
    # BM25 检索器
    bm25_retriever = BM25Retriever.from_documents(documents)
    bm25_retriever.k = 3
    
    # 向量检索器
    vector_retriever = vectorstore.as_retriever(search_kwargs={"k": 3})
    
    # 混合检索器（加权融合）
    ensemble_retriever = EnsembleRetriever(
        retrievers=[bm25_retriever, vector_retriever],
        weights=[0.3, 0.7]  # BM25 权重 0.3，向量权重 0.7
    )
    
    return ensemble_retriever

重排序

from langchain.retrievers.contextual_compression import ContextualCompressionRetriever
from langchain_community.document_transformers import EmbeddingsRedundantFilter

def create_reranker_retriever(base_retriever, embeddings):
    """创建带重排序的检索器"""
    # 使用嵌入冗余过滤器进行重排序
    filter_obj = EmbeddingsRedundantFilter(embeddings=embeddings)
    
    compression_retriever = ContextualCompressionRetriever(
        base_compressor=filter_obj,
        base_retriever=base_retriever
    )
    
    return compression_retriever
