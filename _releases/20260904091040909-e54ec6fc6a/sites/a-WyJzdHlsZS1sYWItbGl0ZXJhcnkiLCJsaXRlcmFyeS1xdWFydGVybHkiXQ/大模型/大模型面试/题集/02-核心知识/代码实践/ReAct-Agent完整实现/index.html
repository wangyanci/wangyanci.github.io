"""
ReAct Agent 完整实现
包含：工具系统、记忆系统、Agent 核心、执行流程
"""

import json
import re
import time
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Callable, Any
from datetime import datetime


# ============== 工具系统 ==============

@dataclass
class ToolResult:
    """工具执行结果"""
    success: bool
    data: Any
    error: Optional[str] = None


class Tool(ABC):
    """工具基类"""
    
    def __init__(self, name: str, description: str):
        self.name = name
        self.description = description
        self.parameters_schema = self._define_parameters()
    
    @abstractmethod
    def _define_parameters(self) -> Dict:
        """定义工具参数 JSON Schema"""
        pass
    
    @abstractmethod
    def execute(self, **kwargs) -> ToolResult:
        """执行工具"""
        pass
    
    def to_function_definition(self) -> Dict:
        """转换为 Function Calling 格式"""
        return {
            "type": "function",
            "function": {
                "name": self.name,
                "description": self.description,
                "parameters": self.parameters_schema
            }
        }


class CalculatorTool(Tool):
    """计算器工具"""
    
    def __init__(self):
        super().__init__(
            name="calculator",
            description="执行数学计算，支持 +, -, *, /, **, % 运算"
        )
    
    def _define_parameters(self) -> Dict:
        return {
            "type": "object",
            "properties": {
                "expression": {
                    "type": "string",
                    "description": "数学表达式，如 '2 + 3 * 4'"
                }
            },
            "required": ["expression"]
        }
    
    def execute(self, expression: str) -> ToolResult:
        try:
            # 安全评估：只允许数学运算
            allowed_names = {
                "abs": abs, "max": max, "min": min,
                "sum": sum, "pow": pow, "round": round
            }
            result = eval(expression, {"__builtins__": {}}, allowed_names)
            return ToolResult(success=True, data=result)
        except Exception as e:
            return ToolResult(success=False, data=None, error=str(e))


class SearchTool(Tool):
    """模拟搜索工具"""
    
    def __init__(self):
        super().__init__(
            name="search",
            description="搜索信息，返回相关结果"
        )
        # 模拟知识库
        self.knowledge_base = {
            "python": "Python 是一种高级编程语言，由 Guido van Rossum 于 1991 年创建。",
            "机器学习": "机器学习是人工智能的一个分支，通过数据训练模型进行预测。",
            "深度学习": "深度学习是机器学习的子集，使用多层神经网络进行学习。",
            "人工智能": "人工智能是计算机科学的一个分支，致力于创建能够执行通常需要人类智能的任务的系统。",
            "Agent": "AI Agent 是能够感知环境、做出决策并执行动作的自主系统。",
        }
    
    def _define_parameters(self) -> Dict:
        return {
            "type": "object",
            "properties": {
                "query": {
                    "type": "string",
                    "description": "搜索关键词"
                }
            },
            "required": ["query"]
        }
    
    def execute(self, query: str) -> ToolResult:
        results = []
        for key, value in self.knowledge_base.items():
            if query.lower() in key.lower() or query.lower() in value.lower():
                results.append(f"{key}: {value}")
        
        if results:
            return ToolResult(success=True, data="\n".join(results))
        return ToolResult(success=False, data=None, error=f"未找到关于 '{query}' 的信息")


class CurrentTimeTool(Tool):
    """获取当前时间工具"""
    
    def __init__(self):
        super().__init__(
            name="get_current_time",
            description="获取当前日期和时间"
        )
    
    def _define_parameters(self) -> Dict:
        return {
            "type": "object",
            "properties": {}
        }
    
    def execute(self) -> ToolResult:
        now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        return ToolResult(success=True, data=now)

2.2 记忆系统

# ============== 记忆系统 ==============

@dataclass
class MemoryEntry:
    """记忆条目"""
    content: str
    timestamp: float
    entry_type: str  # 'thought', 'action', 'observation'


class ShortTermMemory:
    """短期记忆（滑动窗口）"""
    
    def __init__(self, max_entries: int = 20):
        self.entries: List[MemoryEntry] = []
        self.max_entries = max_entries
    
    def add(self, content: str, entry_type: str):
        """添加记忆"""
        entry = MemoryEntry(
            content=content,
            timestamp=time.time(),
            entry_type=entry_type
        )
        self.entries.append(entry)
        
        # 保持窗口大小
        if len(self.entries) > self.max_entries:
            self.entries = self.entries[-self.max_entries:]
    
    def get_history(self) -> str:
        """获取格式化的历史记录"""
        lines = []
        for entry in self.entries:
            prefix = {
                'thought': '🤔 Thought',
                'action': '⚡ Action',
                'observation': '👁️ Observation'
            }.get(entry.entry_type, entry.entry_type)
            lines.append(f"{prefix}: {entry.content}")
        return "\n".join(lines)
    
    def clear(self):
        """清空记忆"""
        self.entries = []


class ToolRegistry:
    """工具注册中心"""
    
    def __init__(self):
        self._tools: Dict[str, Tool] = {}
    
    def register(self, tool: Tool):
        """注册工具"""
        self._tools[tool.name] = tool
    
    def get(self, name: str) -> Optional[Tool]:
        """获取工具"""
        return self._tools.get(name)
    
    def list_tools(self) -> List[Dict]:
        """列出所有工具定义"""
        return [tool.to_function_definition() for tool in self._tools.values()]
    
    def get_tool_descriptions(self) -> str:
        """获取工具描述文本"""
        descriptions = []
        for tool in self._tools.values():
            desc = f"- {tool.name}: {tool.description}"
            params = json.dumps(tool.parameters_schema, ensure_ascii=False)
            desc += f"\n  参数: {params}"
            descriptions.append(desc)
        return "\n".join(descriptions)

# ============== LLM 接口 ==============

class LLMInterface:
    """
    LLM 接口类
    实际项目中应替换为 OpenAI / Claude / 通义千问等真实 API
    """
    
    def __init__(self, model_name: str = "simulated-llm"):
        self.model_name = model_name
    
    def generate(self, prompt: str) -> str:
        """
        生成回复
        实际实现应调用真实 LLM API
        """
        # 这里模拟 LLM 的 ReAct 行为
        # 实际项目中使用:
        # response = openai.chat.completions.create(...)
        
        # 简单的模拟逻辑：解析 prompt 中的模式
        if "Thought:" in prompt and "Action:" not in prompt:
            # 第一轮：生成 Thought + Action
            return self._simulate_first_response(prompt)
        elif "Observation:" in prompt:
            # 后续轮次：基于 Observation 继续
            return self._simulate_follow_up(prompt)
        
        return "Thought: 我已经完成任务\nAction: Finish\nAction Input: {}"
    
    def _simulate_first_response(self, prompt: str) -> str:
        """模拟首轮响应"""
        # 检测用户需求
        if "计算" in prompt or "等于" in prompt or "+" in prompt or "*" in prompt:
            # 提取表达式
            import re
            expr_match = re.search(r'(\d+\s*[+\-*/]+\s*\d+)', prompt)
            if expr_match:
                expr = expr_match.group(1).replace(" ", "")
                return f"Thought: 用户需要计算 {expr}，我应该使用计算器工具\nAction: calculator\nAction Input: {{\"expression\": \"{expr}\"}}"
        
        if "时间" in prompt or "几点" in prompt or "日期" in prompt:
            return "Thought: 用户询问当前时间，我应该查询当前时间\nAction: get_current_time\nAction Input: {}"
        
        if "搜索" in prompt or "什么是" in prompt or "介绍" in prompt:
            # 提取查询词
            query = prompt.split("\n")[-1].replace("现在请回答：", "").strip()
            return f"Thought: 用户想了解 {query}，我应该搜索相关信息\nAction: search\nAction Input: {{\"query\": \"{query}\"}}"
        
        return "Thought: 我已经获得足够信息\nAction: Finish\nAction Input: {\"answer\": \"这是模拟回复。实际项目中请接入真实 LLM API。\"}"
    
    def _simulate_follow_up(self, prompt: str) -> str:
        """模拟后续响应"""
        # 检查是否有 Observation
        if "Observation:" in prompt:
            # 获取最后一个 Observation
            obs_match = re.findall(r'Observation: (.+)', prompt)
            if obs_match:
                observation = obs_match[-1]
                return f"Thought: 工具返回了结果：{observation}。我已经获得足够信息来回答用户。\nAction: Finish\nAction Input: {{\"answer\": \"根据查询结果：{observation}\"}}"
        
        return "Thought: 我已经完成任务\nAction: Finish\nAction Input: {\"answer\": \"任务完成\"}"

2.4 ReAct Agent 核心

# ============== ReAct Agent 核心 ==============

@dataclass
class AgentConfig:
    """Agent 配置"""
    max_iterations: int = 10
    max_retries: int = 3
    verbose: bool = True


class ReActAgent:
    """
    ReAct Agent 完整实现
    
    实现 Thought-Action-Observation 循环
    """
    
    def __init__(
        self,
        llm: LLMInterface,
        tool_registry: ToolRegistry,
        memory: ShortTermMemory,
        config: AgentConfig = None
    ):
        self.llm = llm
        self.tools = tool_registry
        self.memory = memory
        self.config = config or AgentConfig()
    
    def _build_system_prompt(self) -> str:
        """构建系统提示"""
        return """你是一位智能助手，可以使用工具来帮助用户完成任务。

## 可用工具
{tools}

## 指令
请按照以下格式进行思考和行动：

Thought: 你对当前情况的思考，分析用户需求，决定下一步行动
Action: 工具名称（从可用工具中选择，或 Finish 表示完成任务）
Action Input: 工具参数（JSON格式）

当工具返回结果后，你会看到 Observation。
根据 Observation，继续思考并决定下一步。

当任务完成时，使用：
Action: Finish
Action Input: {{"answer": "你的最终回答"}}

## 重要规则
1. 每次只能调用一个工具
2. 必须严格遵循 Thought-Action-Action Input 格式
3. 如果工具调用失败，尝试其他方式
4. 回答问题时必须基于工具返回的结果
"""
    
    def _parse_response(self, response: str) -> Dict[str, str]:
        """
        解析 LLM 响应
        提取 Thought, Action, Action Input
        """
        result = {
            'thought': '',
            'action': '',
            'action_input': '{}'
        }
        
        # 提取 Thought
        thought_match = re.search(r'Thought:\s*(.+?)(?=\nAction:|$)', response, re.DOTALL)
        if thought_match:
            result['thought'] = thought_match.group(1).strip()
        
        # 提取 Action
        action_match = re.search(r'Action:\s*(\w+)', response)
        if action_match:
            result['action'] = action_match.group(1).strip()
        
        # 提取 Action Input
        input_match = re.search(r'Action Input:\s*(\{.*?\})', response, re.DOTALL)
        if input_match:
            result['action_input'] = input_match.group(1).strip()
        
        return result
    
    def _execute_tool(self, action: str, action_input: str) -> str:
        """执行工具"""
        try:
            # 解析参数
            params = json.loads(action_input) if action_input else {}
            
            # 获取工具
            tool = self.tools.get(action)
            if not tool:
                return f"错误: 未找到工具 '{action}'"
            
            # 执行工具
            result = tool.execute(**params)
            
            if result.success:
                return str(result.data)
            else:
                return f"错误: {result.error}"
                
        except json.JSONDecodeError:
            return f"错误: 无效的 JSON 参数: {action_input}"
        except Exception as e:
            return f"错误: 工具执行异常: {str(e)}"
    
    def run(self, user_query: str) -> str:
        """
        执行 ReAct 循环
        
        Args:
            user_query: 用户查询
            
        Returns:
            最终回答
        """
        if self.config.verbose:
            print(f"\n{'='*60}")
            print(f"🚀 开始处理任务: {user_query}")
            print(f"{'='*60}\n")
        
        # 清空记忆
        self.memory.clear()
        
        # 构建初始 prompt
        system_prompt = self._build_system_prompt().format(
            tools=self.tools.get_tool_descriptions()
        )
        
        prompt = f"""{system_prompt}

## 当前任务
{user_query}

开始！
Thought:"""
        
        # ReAct 循环
        for iteration in range(self.config.max_iterations):
            if self.config.verbose:
                print(f"\n--- 迭代 {iteration + 1} ---")
            
            # 1. 调用 LLM 生成 Thought + Action
            response = self.llm.generate(prompt)
            parsed = self._parse_response(response)
            
            thought = parsed['thought']
            action = parsed['action']
            action_input = parsed['action_input']
            
            if self.config.verbose:
                print(f"🤔 Thought: {thought}")
                print(f"⚡ Action: {action}")
                print(f"📥 Input: {action_input}")
            
            # 存储到记忆
            self.memory.add(thought, 'thought')
            self.memory.add(f"{action} {action_input}", 'action')
            
            # 2. 检查是否完成
            if action == "Finish":
                try:
                    finish_data = json.loads(action_input)
                    answer = finish_data.get("answer", "任务完成")
                except:
                    answer = action_input
                
                if self.config.verbose:
                    print(f"\n✅ 任务完成!")
                    print(f"📝 最终答案: {answer}")
                
                return answer
            
            # 3. 执行工具
            observation = self._execute_tool(action, action_input)
            
            if self.config.verbose:
                print(f"👁️ Observation: {observation}")
            
            # 存储 Observation
            self.memory.add(observation, 'observation')
            
            # 4. 更新 prompt
            prompt += f" {thought}\nAction: {action}\nAction Input: {action_input}\n"
            prompt += f"Observation: {observation}\n\nThought:"
        
        # 超过最大迭代次数
        if self.config.verbose:
            print(f"\n⚠️ 达到最大迭代次数 ({self.config.max_iterations})")
        
        return "任务处理超时，请尝试简化问题。"

2.5 主程序入口

# ============== 主程序 ==============

def main():
    """主程序"""
    print("🤖 ReAct Agent 演示")
    print("=" * 60)
    
    # 1. 创建 LLM 接口
    llm = LLMInterface()
    
    # 2. 注册工具
    registry = ToolRegistry()
    registry.register(CalculatorTool())
    registry.register(SearchTool())
    registry.register(CurrentTimeTool())
    
    # 3. 创建记忆
    memory = ShortTermMemory(max_entries=20)
    
    # 4. 创建 Agent
    agent = ReActAgent(
        llm=llm,
        tool_registry=registry,
        memory=memory,
        config=AgentConfig(max_iterations=10, verbose=True)
    )
    
    # 5. 运行示例
    test_queries = [
        "计算 25 + 36 * 2",
        "现在几点了？",
        "搜索 Python 的信息",
    ]
    
    for query in test_queries:
        result = agent.run(query)
        print(f"\n{'='*60}")
        print(f"❓ 问题: {query}")
        print(f"✅ 回答: {result}")
        print(f"{'='*60}\n")


if __name__ == "__main__":
    main()

四、扩展到真实 LLM

import openai

class OpenAILLM(LLMInterface):
    """OpenAI LLM 实现"""
    
    def __init__(self, api_key: str, model: str = "gpt-4"):
        super().__init__(model)
        self.client = openai.OpenAI(api_key=api_key)
        self.model = model
    
    def generate(self, prompt: str) -> str:
        """调用 OpenAI API"""
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": "你是一个使用 ReAct 框架的 AI Agent。"},
                {"role": "user", "content": prompt}
            ],
            temperature=0.7,
            max_tokens=500
        )
        return response.choices[0].message.content
