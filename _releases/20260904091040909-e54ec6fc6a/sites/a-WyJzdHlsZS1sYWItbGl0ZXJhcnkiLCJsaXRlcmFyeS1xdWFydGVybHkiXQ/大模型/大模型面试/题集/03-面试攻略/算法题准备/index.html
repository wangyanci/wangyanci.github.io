Beam Search 实现

def beam_search_decode(logits_fn, start_token, end_token, 
                       beam_width=3, max_length=20):
    """
    Beam Search 解码
    
    Args:
        logits_fn: 输入序列，输出下一个 token 的 logit 分布
        start_token: 起始 token
        end_token: 结束 token
        beam_width: 束宽
        max_length: 最大长度
    
    Returns:
        top-k 序列及其分数
    """
    # 初始化: [([tokens], score)]
    beams = [([start_token], 0.0)]
    completed = []
    
    for _ in range(max_length):
        candidates = []
        
        for seq, score in beams:
            if seq[-1] == end_token:
                completed.append((seq, score))
                continue
            
            # 获取下一个 token 的概率分布
            logits = logits_fn(seq)
            log_probs = np.log(softmax(logits))
            
            # 取 top-k
            top_k_indices = np.argsort(log_probs)[-beam_width:]
            
            for idx in top_k_indices:
                new_seq = seq + [idx]
                new_score = score + log_probs[idx]
                candidates.append((new_seq, new_score))
        
        # 选择 top beam_width 个候选
        candidates.sort(key=lambda x: x[1], reverse=True)
        beams = candidates[:beam_width]
        
        if not beams:
            break
    
    # 合并已完成的序列
    all_results = completed + beams
    all_results.sort(key=lambda x: x[1] / len(x[0]), reverse=True)
    
    return all_results[:beam_width]

Softmax 和 Temperature

import numpy as np

def softmax_with_temperature(logits, temperature=1.0):
    """
    带 Temperature 的 Softmax
    
    Temperature 作用:
    - T → 0: 趋于 argmax (确定性)
    - T = 1: 标准 softmax
    - T → ∞: 趋于均匀分布 (随机性)
    """
    scaled_logits = logits / temperature
    exp_logits = np.exp(scaled_logits - np.max(scaled_logits))
    return exp_logits / np.sum(exp_logits)

2. 文本处理算法

Token 截断（滑动窗口）

def sliding_window_tokenize(text, max_tokens=512, overlap=50):
    """
    滑动窗口分块
    
    将长文本分割为多个有重叠的块
    """
    tokens = text.split()  # 简化版，实际用 Tokenizer
    chunks = []
    
    start = 0
    while start < len(tokens):
        end = min(start + max_tokens, len(tokens))
        chunk = tokens[start:end]
        chunks.append(" ".join(chunk))
        
        if end == len(tokens):
            break
        start = end - overlap
    
    return chunks

最长公共子序列（LCS）- 用于文本相似度

def longest_common_subsequence(text1, text2):
    """
    最长公共子序列
    
    用于评估生成文本与参考文本的相似度
    """
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i-1] == text2[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    
    # 回溯获取 LCS
    lcs = []
    i, j = m, n
    while i > 0 and j > 0:
        if text1[i-1] == text2[j-1]:
            lcs.append(text1[i-1])
            i -= 1
            j -= 1
        elif dp[i-1][j] > dp[i][j-1]:
            i -= 1
        else:
            j -= 1
    
    return "".join(reversed(lcs)), dp[m][n]

余弦相似度

import numpy as np

def cosine_similarity(v1, v2):
    """
    计算两个向量的余弦相似度
    
    返回值范围: [-1, 1]
    """
    dot_product = np.dot(v1, v2)
    norm_v1 = np.linalg.norm(v1)
    norm_v2 = np.linalg.norm(v2)
    
    if norm_v1 == 0 or norm_v2 == 0:
        return 0.0
    
    return dot_product / (norm_v1 * norm_v2)


def top_k_similarity(query_vector, document_vectors, k=5):
    """
    找出与查询向量最相似的 k 个文档
    
    Args:
        query_vector: 查询向量
        document_vectors: 文档向量列表 [(doc_id, vector), ...]
        k: 返回数量
    
    Returns:
        top-k 文档及其相似度分数
    """
    similarities = []
    
    for doc_id, doc_vector in document_vectors:
        sim = cosine_similarity(query_vector, doc_vector)
        similarities.append((doc_id, sim))
    
    # 按相似度降序排列
    similarities.sort(key=lambda x: x[1], reverse=True)
    
    return similarities[:k]

BFS 树搜索（ToT 核心）

from collections import deque

def bfs_tree_search(root, evaluate_fn, beam_width=3, max_depth=5):
    """
    BFS 树搜索 - ToT 的核心算法
    
    Args:
        root: 根节点
        evaluate_fn: 节点评估函数
        beam_width: 每层保留的节点数
        max_depth: 最大搜索深度
    
    Returns:
        最优路径
    """
    current_level = [root]
    
    for depth in range(max_depth):
        next_level = []
        
        for node in current_level:
            # 扩展子节点
            children = node.expand()
            next_level.extend(children)
        
        if not next_level:
            break
        
        # 评估并排序
        scored = [(node, evaluate_fn(node)) for node in next_level]
        scored.sort(key=lambda x: x[1], reverse=True)
        
        # 保留 top-k
        current_level = [node for node, _ in scored[:beam_width]]
        
        # 检查是否到达终止节点
        for node in current_level:
            if node.is_terminal():
                return node.get_path()
    
    # 返回最优路径
    best = max(current_level, key=evaluate_fn)
    return best.get_path()

5. 缓存设计（LRU）

from collections import OrderedDict

class LRUCache:
    """
    LRU 缓存 - 用于 Agent 记忆缓存
    """
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()
    
    def get(self, key: str):
        if key not in self.cache:
            return None
        # 移动到末尾（最近使用）
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key: str, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        
        if len(self.cache) > self.capacity:
            # 移除最久未使用的
            self.cache.popitem(last=False)
    
    def get_all(self):
        """获取所有缓存项"""
        return list(self.cache.items())
