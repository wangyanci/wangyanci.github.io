生产级 Agent 设计

一、生产级 Agent 的核心要求

生产环境的 Agent 与原型/demo 有本质区别，需要在可靠性、可观测性、安全性、性能四个维度满足企业级要求。

二、状态机设计

为什么需要状态机？

Agent 不是无状态的函数调用，而是有状态的会话系统。状态机确保 Agent 在任何时刻都知道"自己在哪里"和"下一步该做什么"。

状态机代码实现

from enum import Enum, auto
from dataclasses import dataclass
from typing import Optional

class AgentState(Enum):
    """Agent 状态枚举"""
    INIT = auto()
    COLLECTING_PARAMS = auto()
    REASONING = auto()
    TOOL_CALLING = auto()
    GENERATING_RESPONSE = auto()
    OUTPUT_CHECKING = auto()
    RETRYING = auto()
    ERROR_HANDLING = auto()
    COMPLETED = auto()
    CANCELLED = auto()

@dataclass
class StateContext:
    """状态上下文"""
    state: AgentState
    attempt_count: int = 0
    max_retries: int = 3
    collected_params: dict = None
    tool_results: list = None
    error: Optional[Exception] = None

class AgentStateMachine:
    """Agent 状态机"""
    
    def __init__(self):
        self.transitions = {
            AgentState.INIT: [
                AgentState.COLLECTING_PARAMS,
                AgentState.REASONING,
                AgentState.ERROR_HANDLING
            ],
            AgentState.COLLECTING_PARAMS: [
                AgentState.COLLECTING_PARAMS,
                AgentState.REASONING,
                AgentState.CANCELLED
            ],
            AgentState.REASONING: [
                AgentState.TOOL_CALLING,
                AgentState.GENERATING_RESPONSE,
                AgentState.ERROR_HANDLING,
                AgentState.RETRYING
            ],
            AgentState.TOOL_CALLING: [
                AgentState.TOOL_CALLING,
                AgentState.REASONING,
                AgentState.ERROR_HANDLING
            ],
            AgentState.GENERATING_RESPONSE: [
                AgentState.OUTPUT_CHECKING
            ],
            AgentState.OUTPUT_CHECKING: [
                AgentState.COMPLETED,
                AgentState.RETRYING
            ],
            AgentState.RETRYING: [
                AgentState.REASONING,
                AgentState.ERROR_HANDLING
            ],
            AgentState.ERROR_HANDLING: [
                AgentState.COMPLETED
            ],
            AgentState.CANCELLED: [
                AgentState.COMPLETED
            ]
        }
        self.context = StateContext(state=AgentState.INIT)
    
    def can_transition(self, new_state: AgentState) -> bool:
        """检查状态转换是否合法"""
        valid_states = self.transitions.get(self.context.state, [])
        return new_state in valid_states
    
    def transition(self, new_state: AgentState) -> bool:
        """执行状态转换"""
        if not self.can_transition(new_state):
            raise InvalidStateTransition(
                f"Cannot transition from {self.context.state} to {new_state}"
            )
        
        old_state = self.context.state
        self.context.state = new_state
        
        # 状态转换日志
        logger.info(f"State transition: {old_state.name} -> {new_state.name}")
        
        # 状态转换钩子
        self._on_state_enter(new_state)
        return True
    
    def _on_state_enter(self, state: AgentState):
        """状态进入时的处理"""
        if state == AgentState.RETRYING:
            self.context.attempt_count += 1
            if self.context.attempt_count > self.context.max_retries:
                self.transition(AgentState.ERROR_HANDLING)

容错代码实现

import asyncio
from functools import wraps
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"      # 正常
    OPEN = "open"          # 熔断
    HALF_OPEN = "half_open"  # 半开测试

class CircuitBreaker:
    """熔断器"""
    
    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: int = 60,
        half_open_max_calls: int = 3
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_calls = half_open_max_calls
        
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = None
        self.half_open_calls = 0
    
    async def call(self, func, *args, **kwargs):
        """带熔断保护的调用"""
        if self.state == CircuitState.OPEN:
            if self._should_attempt_reset():
                self.state = CircuitState.HALF_OPEN
                self.half_open_calls = 0
            else:
                raise CircuitBreakerOpen("服务暂时不可用，请稍后重试")
        
        if self.state == CircuitState.HALF_OPEN:
            if self.half_open_calls >= self.half_open_max_calls:
                raise CircuitBreakerOpen("半开测试进行中")
            self.half_open_calls += 1
        
        try:
            result = await func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise
    
    def _on_success(self):
        """成功处理"""
        self.failure_count = 0
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.CLOSED
            self.half_open_calls = 0
    
    def _on_failure(self):
        """失败处理"""
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
    
    def _should_attempt_reset(self) -> bool:
        """检查是否应该尝试恢复"""
        if self.last_failure_time is None:
            return True
        return (time.time() - self.last_failure_time) > self.recovery_timeout


class RetryWithBackoff:
    """带指数退避的重试"""
    
    def __init__(
        self,
        max_retries: int = 3,
        base_delay: float = 1.0,
        max_delay: float = 60.0,
        exponential_base: float = 2.0
    ):
        self.max_retries = max_retries
        self.base_delay = base_delay
        self.max_delay = max_delay
        self.exponential_base = exponential_base
    
    async def execute(self, func, *args, **kwargs):
        """执行带重试的函数"""
        for attempt in range(self.max_retries + 1):
            try:
                return await func(*args, **kwargs)
            except Exception as e:
                if attempt == self.max_retries:
                    raise
                
                delay = min(
                    self.base_delay * (self.exponential_base ** attempt),
                    self.max_delay
                )
                logger.warning(
                    f"Attempt {attempt + 1} failed: {e}. "
                    f"Retrying in {delay}s..."
                )
                await asyncio.sleep(delay)

四、降级策略

降级实现代码

class DegradationStrategy:
    """降级策略管理"""
    
    def __init__(self):
        self.levels = [
            FullCapability(),      # 正常
            ReducedModel(),        # 降级1
            CoreToolsOnly(),       # 降级2
            RuleBasedFallback(),   # 降级3
            StaticResponse()       # 降级4
        ]
        self.current_level = 0
    
    async def execute(self, task: Task) -> Response:
        """按当前降级级别执行"""
        for level in range(self.current_level, len(self.levels)):
            strategy = self.levels[level]
            
            if await strategy.is_available():
                try:
                    result = await strategy.execute(task)
                    logger.info(f"Executed at degradation level {level}")
                    return result
                except Exception as e:
                    logger.error(f"Level {level} failed: {e}")
                    continue
        
        return Response(
            content="系统暂时不可用，请稍后重试",
            status="error"
        )
    
    def escalate(self):
        """升级降级级别"""
        if self.current_level < len(self.levels) - 1:
            self.current_level += 1
            logger.warning(f"Degradation escalated to level {self.current_level}")
    
    def recover(self):
        """尝试恢复"""
        if self.current_level > 0:
            self.current_level -= 1
            logger.info(f"Degradation recovered to level {self.current_level}")

六、常见面试考点

Q1: 生产级 Agent 需要哪些核心能力？

答: 四个方面：可靠性（状态机、容错、降级）、可观测性（日志、指标、追踪）、安全性（输入输出检查、数据保护）、性能（延迟、并发、资源）。

Q2: 熔断器的工作原理？

答: 熔断器有三种状态：Closed（正常）、Open（熔断，拒绝请求）、Half-Open（半开，允许少量请求测试）。当失败次数超过阈值时打开，经过一段时间后进入半开状态测试恢复。

Q3: 降级策略如何设计？

答: 设计多层降级：正常→简化模型→减少工具→规则引擎→静态响应。每层确保比上层更可靠但功能更少，形成"优雅降级"的漏斗。
